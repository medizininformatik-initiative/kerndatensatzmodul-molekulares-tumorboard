// ---------------------------------------------------------------------------
// Root group: iterate over the whole PatientRecord and populate the Bundle
// ---------------------------------------------------------------------------
group SpecimenHistologyIHCMSI(source src : DNPMPatientRecord, target bundle : Bundle) {
  // Bundle type = collection
  src -> bundle.type = 'collection' "bundleType";
  // --- Specimen (one per specimens entry) -----------------------------------
  src.specimens as spec ->  bundle.entry as e,  e.resource = create('Specimen') as s then {
    spec.id as specid -> e.fullUrl = append('urn:uuid:', specid) "fullUrl";
    spec -> s then MapSpecimen(spec, s) "map";
  } "specimen";
  // --- DiagnosticReport + child Observations per histologyReport -----------
  src.histologyReports as hr ->  bundle.entry as e,  e.resource = create('DiagnosticReport') as dr then {
    hr.id as hrid -> e.fullUrl = append('urn:uuid:', hrid) "fullUrl";
    hr -> dr then MapHistologyReport(hr, dr, bundle) "map";
  } "histologyReport";
  // --- DiagnosticReport + child Observations per ihcReport -----------------
  src.ihcReports as ihc ->  bundle.entry as e,  e.resource = create('DiagnosticReport') as dr then {
    ihc.id as ihcid -> e.fullUrl = append('urn:uuid:', ihcid) "fullUrl";
    ihc -> dr then MapIHCReport(ihc, dr, bundle) "map";
  } "ihcReport";
  // --- Observation per msi finding -----------------------------------------
  src.msiFindings as msi ->  bundle.entry as e,  e.resource = create('Observation') as obs then {
    msi.id as msiid -> e.fullUrl = append('urn:uuid:', msiid) "fullUrl";
    msi -> obs then MapMSIFinding(msi, obs) "map";
  } "msiFinding";
}

// ---------------------------------------------------------------------------
// MapSpecimen: DNPM specimen entry → FHIR Specimen
// Profile: MII (parent: Specimen)
// ---------------------------------------------------------------------------
group MapSpecimen(source src : BackboneElement, target tgt : Specimen) {
  src.id as srcId -> tgt.id = cast(srcId, 'id') "specimenId";
  // Profile meta
  src -> tgt.meta as m then {
    src -> m.profile = 'https://www.medizininformatik-initiative.de/fhir/ext/modul-mtb/StructureDefinition/mii-pr-mtb-tumor-specimen' "specimenProfile";
  } "specimenMeta";
  // subject → Patient reference
  src.patient as pat -> tgt.subject as subj then {
    pat.id as pid -> subj.reference = append('urn:uuid:', pid) "specimenSubjectRef";
  } "specimenSubject";
  // type → CodeableConcept
  src.type as t -> tgt.type as typ then {
    t -> typ.coding as c then {
      t.code as codeV -> c.code = cast(codeV, 'string') "specimenTypeCode";
      t.system as systemV -> c.system = cast(systemV, 'string') "specimenTypeSystem";
    } "specimenTypeCoding";
  } "specimenType";
  // collection block
  src.collection as col -> tgt.collection as tcol then MapSpecimenCollection(col, tcol) "specimenCollection";
}

// MapHistologyReport: DNPM histologyReport → FHIR DiagnosticReport
// Also creates child Observation entries for tumorCellContent and
// tumorMorphology and links them as DiagnosticReport.result references.
// ---------------------------------------------------------------------------
group MapHistologyReport(source src : BackboneElement, target dr : DiagnosticReport, target bundle : Bundle) {
  src.id as idV -> dr.id = cast(idV, 'string') "drId";
  // Profile meta
  src -> dr.meta as m then {
    src -> m.profile = 'https://www.medizininformatik-initiative.de/fhir/ext/modul-mtb/StructureDefinition/mii-pr-mtb-histologie-befundbericht' "drProfile";
  } "drMeta";
  src -> dr.status = 'final' "drStatus";
  // subject
  src.patient as pat -> dr.subject as subj then {
    pat.id as pid -> subj.reference = append('urn:uuid:', pid) "drSubjectRef";
  } "drSubject";
  // specimen[0]
  src.specimen as spec -> dr.specimen as sRef then {
    spec.id as sid -> sRef.reference = append('urn:uuid:', sid) "drSpecimenRef";
  } "drSpecimen";
  // issued — append timestamp suffix so it becomes a FHIR instant
  src.issuedOn as d -> dr.issued = append(d, 'T00:00:00Z') "drIssued";
  // --- tumorCellContent child Observation -----------------------------------
  src.results as res ->  bundle.entry as e,  e.resource = create('Observation') as obs then {
    res.tumorCellContent as tcc then {
      tcc.id as tid -> e.fullUrl = append('urn:uuid:', tid) "fullUrl";
      tcc -> obs then MapTumorCellContent(tcc, obs) "tccMap";
    } "tccInner";
  } "tccEntry";
  // link result reference from DiagnosticReport
  src.results as res -> dr.result as r then {
    res.tumorCellContent as tcc then {
      tcc.id as tid -> r.reference = append('urn:uuid:', tid) "tccRef";
    } "tccInner";
  } "drResultTCC";
  // --- tumorMorphology child Observation ------------------------------------
  src.results as res ->  bundle.entry as e,  e.resource = create('Observation') as obs then {
    res.tumorMorphology as tm then {
      tm.id as tmid -> e.fullUrl = append('urn:uuid:', tmid) "fullUrl";
      tm -> obs then MapTumorMorphology(tm, obs) "morphMap";
    } "morphInner";
  } "morphEntry";
  // link result reference from DiagnosticReport
  src.results as res -> dr.result as r then {
    res.tumorMorphology as tm then {
      tm.id as tmid -> r.reference = append('urn:uuid:', tmid) "morphRef";
    } "morphInner";
  } "drResultMorph";
}
