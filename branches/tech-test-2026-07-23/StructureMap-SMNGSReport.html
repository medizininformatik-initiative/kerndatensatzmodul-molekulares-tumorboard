// ---------------------------------------------------------------------------
// Root group: iterate over all ngsReports in the PatientRecord
// ---------------------------------------------------------------------------
group NGSReports(source src : DNPMPatientRecord, target bundle : Bundle) {
  src -> bundle.type = 'collection' "bundleType";
  src.ngsReports as ngs ->  bundle.entry as e,  e.resource = create('DiagnosticReport') as dr then {
    ngs.id as ngsid -> e.fullUrl = append('urn:uuid:', ngsid) "fullUrl";
    ngs -> dr then MapNGSReport(ngs, dr, bundle) "map";
  } "ngsReport";
}

// ---------------------------------------------------------------------------
// MapNGSReport: one DNPM NGS report → DiagnosticReport + child Observations
// ---------------------------------------------------------------------------
group MapNGSReport(source src : BackboneElement, target tgt : DiagnosticReport, target bundle : Bundle) {
  // ---- DiagnosticReport header ----
  src.id as srcId -> tgt.id = cast(srcId, 'id') "drId";
  src -> tgt.status = 'final' "drStatus";
  // Profile declaration
  src -> tgt.meta as m then {
    src -> m.profile = 'https://www.medizininformatik-initiative.de/fhir/ext/modul-mtb/StructureDefinition/mii-pr-mtb-ngs-bericht' "drProfile";
  } "drMeta";
  // Subject
  src.patient as pat -> tgt.subject as subj then {
    pat.id as pid -> subj.reference = append('urn:uuid:', pid) "drSubjectRef";
  } "drSubject";
  // Specimen
  src.specimen as spec -> tgt.specimen as s then {
    spec.id as sid -> s.reference = append('urn:uuid:', sid) "drSpecimenRef";
  } "drSpecimen";
  // Issued (date → instant)
  src.issuedOn as d -> tgt.issued = append(d, 'T00:00:00Z') "drIssued";
  // ---- NGS type → code ----
  src.type as t -> tgt.code as c then {
    t -> c.coding as coding then {
      t.code as codeV -> coding.code = cast(codeV, 'string') "typeCode";
      t.system as systemV -> coding.system = cast(systemV, 'string') "typeSystem";

  // =========================================================================
  // TMB (Mutationslast)
  // =========================================================================
  src.results as r -> bundle then {
    r.tmb as tmb ->  bundle.entry as e,  e.resource = create('Observation') as obs then {
      tmb.id as tmbid -> e.fullUrl = append('urn:uuid:', tmbid) "fullUrl";
      tmb -> obs then MapTMB(tmb, obs) "map";
    } "tmbEntry";
  } "tmbResults";
  // Reference from DiagnosticReport.result
  src.results as r -> tgt then {
    r.tmb as tmb -> tgt.result as res then {
      tmb.id as tid -> res.reference = append('urn:uuid:', tid) "tmbResultRef";

  // =========================================================================
  // BRCAness
  // =========================================================================
  src.results as r -> bundle then {

    r.brcaness as brca -> tgt.result as res then {
      brca.id as bid -> res.reference = append('urn:uuid:', bid) "brcaResultRef";
    } "brcaResult";
  } "brcaResultsRef";
  // =========================================================================
  // HRD Score
  // =========================================================================
  src.results as r -> bundle then {
    r.hrdScore as hrd ->  bundle.entry as e,  e.resource = create('Observation') as obs then {
      hrd.id as hrdid -> e.fullUrl = append('urn:uuid:', hrdid) "fullUrl";
      hrd -> obs then MapHRDScore(hrd, obs) "map";
    } "hrdEntry";
  } "hrdResults";
  src.results as r -> tgt then {
    r.hrdScore as hrd -> tgt.result as res then {
      hrd.id as hid -> res.reference = append('urn:uuid:', hid) "hrdResultRef";
    } "hrdResult";
  } "hrdResultsRef";
  // =========================================================================
  // Simple Variants (SNVs / Indels)
  // =========================================================================
  src.results as r -> bundle then {
    r.simpleVariants as sv ->  bundle.entry as e,  e.resource = create('Observation') as obs then {
      sv.id as svid -> e.fullUrl = append('urn:uuid:', svid) "fullUrl";
      sv -> obs then MapSimpleVariant(sv, obs, src) "map";
    } "svEntry";
  } "svResults";
  src.results as r -> tgt then {
    r.simpleVariants as sv -> tgt.result as res then {
      sv.id as svid -> res.reference = append('urn:uuid:', svid) "svResultRef";
    } "svResult";
  } "svResultsRef";
  // =========================================================================
  // Copy Number Variants
  // =========================================================================
  src.results as r -> bundle then {
    r.copyNumberVariants as cnv ->  bundle.entry as e,  e.resource = create('Observation') as obs then {
      cnv.id as cnvid -> e.fullUrl = append('urn:uuid:', cnvid) "fullUrl";

  // =========================================================================
  // DNA Fusions
  // =========================================================================
  src.results as r -> bundle then {
    r.dnaFusions as df ->  bundle.entry as e,  e.resource = create('Observation') as obs then {
      df.id as dfid -> e.fullUrl = append('urn:uuid:', dfid) "fullUrl";

    } "dfEntry";
  } "dfResults";

      df.id as did -> res.reference = append('urn:uuid:', did) "dfResultRef";
    } "dfResult";
  } "dfResultsRef";
  // =========================================================================
  // RNA Fusions
  // =========================================================================
  src.results as r -> bundle then {
    r.rnaFusions as rf ->  bundle.entry as e,  e.resource = create('Observation') as obs then {
      rf.id as rfid -> e.fullUrl = append('urn:uuid:', rfid) "fullUrl";
      rf -> obs then MapRNAFusion(rf, obs, src) "map";
    } "rfEntry";
  } "rfResults";
  src.results as r -> tgt then {
    r.rnaFusions as rf -> tgt.result as res then {
      rf.id as rid -> res.reference = append('urn:uuid:', rid) "rfResultRef";
    } "rfResult";
  } "rfResultsRef";
}
