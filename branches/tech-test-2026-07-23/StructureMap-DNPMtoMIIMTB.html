group DNPMPatientRecordToBundle(source src : DNPMPatientRecord, target bundle : Bundle) {
  // Set bundle type
  src -> bundle.type = 'transaction' "bundleType";
  // Patient demographics, episodes, family history
  src -> bundle then PatientDemographics(src, bundle) "demographics";
  // Diagnoses + ECOG
  src -> bundle then Diagnoses(src, bundle) "diagnoses";
  // Specimens, histology, IHC, MSI
  src -> bundle then SpecimenHistologyIHCMSI(src, bundle) "pathology";
  // NGS Reports
  src -> bundle then NGSReports(src, bundle) "ngs";
  // CarePlans + recommendations
  src -> bundle then CarePlans(src, bundle) "carePlans";
  // Guideline therapies + procedures
  src -> bundle then GuidelineTherapy(src, bundle) "guidelineTherapy";
  // Follow-ups
  src -> bundle then FollowUps(src, bundle) "followUps";
  // Systemic therapies + responses
  src -> bundle then SystemicTherapies(src, bundle) "systemicTherapies";
  // Claims + claim responses
  src -> bundle then Claims(src, bundle) "claims";
}
